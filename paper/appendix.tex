\section{Wasserstein Distance}
\label{app:wasserstein}

\begin{proposition}
    Wasserstein distance is superior to MSE and KL divergence for financial returns due to three properties: metric structure, tail sensitivity, and geometric interpretation.
\end{proposition}

\begin{proof}

\textit{Property 1 (Metric Structure).} The 1-Wasserstein distance satisfies:
\begin{enumerate}
    \item Non-negativity: $\mathcal{W}_1(P, Q) \geq 0$
    \item Identity: $\mathcal{W}_1(P, Q) = 0 \iff P = Q$
    \item Symmetry: $\mathcal{W}_1(P, Q) = \mathcal{W}_1(Q, P)$
    \item Triangle inequality: $\mathcal{W}_1(P, R) \leq \mathcal{W}_1(P, Q) + \mathcal{W}_1(Q, R)$
\end{enumerate}
KL divergence violates symmetry and triangle inequality, making it unsuitable as a training objective.

\textit{Property 2 (Tail Sensitivity).} For empirical distributions with ordered samples, the Wasserstein distance is:
\begin{equation}
\mathcal{W}_1 = \frac{1}{N} \sum_{i=1}^{N} |x_{(i)} - y_{(i)}|
\end{equation}
This equally weights all quantiles, including extremes. MSE, by contrast, is dominated by the mean: $\text{MSE} = \frac{1}{N}\sum (x_i - y_i)^2$.

\textit{Property 3 (Geometric Interpretation).} Wasserstein distance measures the minimal cost of transporting probability mass from $P$ to $Q$, respecting the metric structure of the underlying space. This is crucial for returns where the magnitude (not just probability) of extreme events matters. $\square$

\end{proof}

\section{Theoretical Properties of the Wasserstein Training Objective}
\label{app:wasserstein_theory}

We analyze three theoretical properties of the 1-Wasserstein reward $r^{\text{rl}} = 1/(1 + \mathcal{W}_1)$: gradient characterization, convergence guarantees, and CVaR control.

\subsection{Gradient Characterization in One Dimension}

For one-dimensional distributions, the 1-Wasserstein distance admits a closed-form representation via CDFs~\cite{villani2009optimal}:
\begin{align}
\mathcal{W}_1(P, Q) &= \int_{-\infty}^{\infty} |F_P(x) - F_Q(x)| \, dx \\
&= \int_{0}^{1} |F_P^{-1}(u) - F_Q^{-1}(u)| \, du
\end{align}
where $F_P^{-1}, F_Q^{-1}$ are quantile functions. The functional derivative with respect to the predicted CDF $F_\theta$ is:
\begin{equation}
\frac{\delta \mathcal{W}_1}{\delta F_\theta}(x) = \text{sgn}(F_\theta(x) - F^*(x)),
\end{equation}
yielding a clear training signal: overestimating the true CDF at $x$ pushes $F_\theta(x)$ downward, and vice versa. For the reward $r^{\text{rl}} = 1/(1+\mathcal{W}_1)$, the gradient:
\begin{equation}
\frac{\partial r^{\text{rl}}}{\partial \theta} = -\frac{1}{(1+\mathcal{W}_1)^2} \cdot \frac{\partial \mathcal{W}_1}{\partial \theta}
\end{equation}
preserves direction with adaptive scaling.

\subsection{Convergence and Sample Complexity}

$\mathcal{W}_1$ metrizes weak convergence on spaces with bounded first moment~\cite{villani2009optimal}. For empirical distributions with $n,m$ i.i.d.\ samples,
\begin{equation}
\mathbb{E}\big[|\mathcal{W}_1(\hat{P}_n, \hat{Q}_m) - \mathcal{W}_1(P,Q)|\big] = O(n^{-1/2} + m^{-1/2}),
\end{equation}
under finite second moment. This contrasts favorably with KL divergence (unbounded under mismatch) and MSE (degraded under heavy tails~\cite{cont2001empirical}).

\subsection{Relationship with Conditional Value-at-Risk (CVaR)}

\begin{proposition}[CVaR Bound via Wasserstein Distance]
Let $P$ and $Q$ be probability distributions on $\mathbb{R}$ with finite first moments. Then for any $\alpha \in (0,1)$:
\begin{equation}
|\text{CVaR}_\alpha(P) - \text{CVaR}_\alpha(Q)|
\leq \frac{1}{1-\alpha} \cdot \mathcal{W}_1(P, Q).
\end{equation}
\end{proposition}

\begin{proof}
From the quantile representation and the definition:
\begin{equation}
\text{CVaR}_\alpha(P) = \frac{1}{1-\alpha} \int_{\alpha}^{1} F_P^{-1}(u) \, du,
\end{equation}
we have:
\begin{align}
|\text{CVaR}_\alpha(P) - \text{CVaR}_\alpha(Q)|
&\leq \frac{1}{1-\alpha} \int_{\alpha}^{1} |F_P^{-1}(u) - F_Q^{-1}(u)| \, du \nonumber \\
&\leq \frac{1}{1-\alpha} \int_{0}^{1} |F_P^{-1}(u) - F_Q^{-1}(u)| \, du \nonumber \\
&= \frac{1}{1-\alpha} \cdot \mathcal{W}_1(P, Q). \quad \square
\end{align}
\end{proof}

For $\alpha = 0.95$, the bound is $20 \cdot \mathcal{W}_1$, so $\mathcal{W}_1 = 0.01$ guarantees $\text{CVaR}_{0.95}$ agreement within 0.2 units. This provides $\alpha$-agnostic tail regularization without explicit CVaR tuning~\cite{chow2015cvar}.

\paragraph{Volatility-Adjusted Momentum}
\begin{multline}
e = \frac{\text{mom}(x_{\text{close}}, 10)}{\text{vol}(x_{\text{close}}, 20)} \\
\quad \times \text{ite}\big(\text{ma}(x_{\text{close}}, 5) > \text{ma}(x_{\text{close}}, 20),\, 1,\, 0\big)
\end{multline}
\textit{Interpretation:} Momentum normalized by volatility, filtered by trend confirmation (golden cross).

\paragraph{Mean Reversion with Volume}
\begin{multline}
e = \big(x_{\text{close}} - \text{ma}(x_{\text{close}}, 20)\big) \\
\quad \times \frac{x_{\text{volume}}}{\text{ma}(x_{\text{volume}}, 20)}
\end{multline}
\textit{Interpretation:} Price deviation scaled by relative volume (higher conviction on high volume).
